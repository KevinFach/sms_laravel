// ════════════════════════════════════════════════════════
//  ANIMACIÓN: LOGO "FAST SMS" CON EFECTO TYPEWRITER
// ════════════════════════════════════════════════════════
void drawLogoScreen() {
  // Fondo negro, texto aparece letra a letra
  String textoBig = "FAST";
  String textoSub = "SMS";

  // Efecto: FAST aparece carácter a carácter
  for (int i = 1; i <= (int)textoBig.length(); i++) {
    display.clearDisplay();
    display.setTextSize(3);
    display.setTextColor(SSD1306_WHITE);
    // Centrar "FAST" → ancho letra size3 = 18px, 4 letras = 72px → x = (128-72)/2 = 28
    display.setCursor(28, 6);
    display.print(textoBig.substring(0, i));
    display.display();
    delay(120);
  }

  delay(200);

  // SMS aparece con efecto inverso (bloque blanco → texto negro)
  for (int i = 1; i <= (int)textoSub.length(); i++) {
    display.setTextSize(3);
    display.setTextColor(SSD1306_WHITE);
    display.setCursor(28, 6);
    display.print(textoBig);

    // Caja rellena detrás de SMS
    int boxW = i * 18 + 4;
    display.fillRect(26, 36, boxW, 24, SSD1306_WHITE);
    display.setTextColor(SSD1306_BLACK);
    display.setCursor(28, 38);
    display.print(textoSub.substring(0, i));
    display.display();
    delay(140);
  }

  delay(300);

  // Parpadeo final del logo completo x3
  for (int p = 0; p < 3; p++) {
    display.invertDisplay(true);
    delay(120);
    display.invertDisplay(false);
    delay(120);
  }

  delay(500);
}

void drawBootScreen(String paso, int progreso) {

  display.setCursor(4, 2);

  // Separador
  display.drawLine(0, 20, 128, 20, SSD1306_WHITE);

  // Paso actual con puntos animados
  int dots = (millis() / 300) % 4;
  String animPaso = paso;
  for (int i = 0; i < dots; i++) animPaso += ".";

  display.setCursor(54, 53);
  display.print(String(progreso) + "%");

  display.display();

  }

// ════════════════════════════════════════════════════════
//  SETUP
// ════════════════════════════════════════════════════════
void setup() {
  Serial.begin(115200);
  log("=============================");
  log("     FAST SMS GATEWAY        ");
  log("=============================");

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    log("ERROR: Pantalla OLED no detectada");
  }

  // ── LOGO ANIMADO ──
  drawLogoScreen();

  // ── BOOT: Modem ──
  drawBootScreen("Iniciando sistema", 10);
  log("Inicializando sistema...");
  delay(400);

  Serial1.begin(MODEM_BAUD, SERIAL_8N1, MODEM_RX, MODEM_TX);
  drawBootScreen("Encendiendo modem", 25);
  log("Encendiendo modem SIM7600E...");

  pinMode(MODEM_PWRKEY, OUTPUT);
  digitalWrite(MODEM_PWRKEY, LOW); delay(100);
  digitalWrite(MODEM_PWRKEY, HIGH); delay(1000);
  digitalWrite(MODEM_PWRKEY, LOW);

  drawBootScreen("Esperando modem", 45);
  log("Esperando respuesta del modem...");
  delay(5000);

  // Verificar modem con AT
  Serial1.println("AT");
  delay(300);
  Serial1.println("AT+CMGF=1");
  delay(500);

  String respAT = "";
  unsigned long t = millis();
  while (millis() - t < 2000) {
    while (Serial1.available()) respAT += (char)Serial1.read();
  }
  modemOK = (respAT.indexOf("OK") != -1);

  if (modemOK) {
    log("Modem OK - Respuesta: " + respAT);
  } else {
    logError("Modem no respondio correctamente. Respuesta: '" + respAT + "'");
  }

  drawBootScreen("Conectando WiFi", 65);
  log("Conectando a WiFi: " + String(ssid));
  conectarWiFi();

  drawBootScreen("Configurando API", 85);
  log("Servidor API: " + String(servidor));
  delay(400);

  drawBootScreen("Listo!", 100);
  log("Sistema listo. Iniciando loop.");
  delay(800);

  ultimoEstado = "En espera";
  drawHUD();
}

// ════════════════════════════════════════════════════════
//  LOOP
// ════════════════════════════════════════════════════════
void loop() {
  // Reconexión WiFi
  if (WiFi.status() != WL_CONNECTED) {
    wifiOK = false;
    log("WiFi desconectado. Reconectando...");
    drawSystemErrorScreen("WiFi perdido", "Reconectando...");
    delay(1500);
    conectarWiFi();
  }

  // Animación idle mientras espera
  if (millis() - ultimaConsulta > (intervaloConsulta - 1000)) {
    drawIdleAnimation();
    delay(200);
  }

  // Consulta API cada intervalo
  if (millis() - ultimaConsulta > intervaloConsulta) {
    ultimoEstado = "Consultando API";
    ultimoError = "";
    drawHUD();
    log("Consultando pendientes...");
    consultarPendientes();
    ultimaConsulta = millis();
    ultimoEstado = "En espera";
    drawHUD();
  }
}

// ════════════════════════════════════════════════════════
//  WIFI
// ════════════════════════════════════════════════════════
void conectarWiFi() {
  WiFi.begin(ssid, password);
  int intentos = 0;
  while (WiFi.status() != WL_CONNECTED && intentos < 20) {
    delay(500);
    intentos++;
    // Actualizar barra de boot si estamos en setup (progreso 65-80)
  }

  if (WiFi.status() == WL_CONNECTED) {
    wifiOK = true;
    log("WiFi conectado. IP: " + WiFi.localIP().toString());
  } else {
    wifiOK = false;
    logError("No se pudo conectar a WiFi '" + String(ssid) + "' tras " + String(intentos) + " intentos");
    drawSystemErrorScreen("WiFi fallido", String(ssid));
    delay(2000);
  }
}

// ════════════════════════════════════════════════════════
//  CONSULTAR API
// ════════════════════════════════════════════════════════
void consultarPendientes() {
  if (WiFi.status() != WL_CONNECTED) {
    logError("Sin WiFi, no se puede consultar la API");
    return;
  }

  HTTPClient http;
  String url = String(servidor) + "/api/pendientes?token=" + token;
  log("GET " + url);

  http.begin(url);
  http.setTimeout(8000);
  int httpCode = http.GET();

  if (httpCode == 200) {
    String respuesta = http.getString();
    respuesta.trim();
    log("Respuesta API (" + String(respuesta.length()) + " bytes)");

    if (respuesta.length() > 5) {
      procesarRespuesta(respuesta);
    } else {
      log("Sin mensajes pendientes.");
    }
  } else if (httpCode > 0) {
    String razon = "HTTP " + String(httpCode);
    if (httpCode == 401) razon = "Token invalido (401)";
    else if (httpCode == 404) razon = "Endpoint no existe (404)";
    else if (httpCode == 500) razon = "Error servidor (500)";
    logError("API respondio: " + razon);
    drawSystemErrorScreen("API Error", razon);
    delay(2000);
  } else {
    String razon = "Sin respuesta / Timeout";
    if (httpCode == HTTPC_ERROR_CONNECTION_REFUSED) razon = "Conexion rechazada";
    else if (httpCode == HTTPC_ERROR_SEND_HEADER_FAILED) razon = "Fallo cabecera HTTP";
    else if (httpCode == HTTPC_ERROR_NOT_CONNECTED)  razon = "No conectado";
    logError("Error HTTP: " + razon + " (code " + String(httpCode) + ")");
    drawSystemErrorScreen("API Sin resp.", razon);
    delay(2000);
  }

  http.end();
}

// ════════════════════════════════════════════════════════
//  PROCESAR RESPUESTA
// ════════════════════════════════════════════════════════
void procesarRespuesta(String respuesta) {
  int startIdx = 0;
  while (startIdx < respuesta.length()) {
    int endIdx = respuesta.indexOf('\n', startIdx);
    String linea = (endIdx == -1) ? respuesta.substring(startIdx) : respuesta.substring(startIdx, endIdx);
    startIdx = (endIdx == -1) ? respuesta.length() : endIdx + 1;

    linea.trim();
    if (linea.length() < 5) continue;

    int p1 = linea.indexOf('|');
    int p2 = linea.indexOf('|', p1 + 1);

    if (p1 == -1 || p2 == -1) {
      logError("Formato incorrecto en linea: " + linea);
      continue;
    }

    String id      = linea.substring(0, p1);
    String numero  = linea.substring(p1 + 1, p2);
    String mensaje = linea.substring(p2 + 1);

    smsTotales++;
    log("Procesando ID:" + id + " | Para:" + numero + " | Msg:" + mensaje);

    // Animación de envío
    unsigned long tSend = millis();
    while (millis() - tSend < 1200) {
      drawSendingScreen(numero, id);
      delay(100);
    }

    String razonFallo = "";
    bool ok = enviarSMS(numero, mensaje, razonFallo);

    if (ok) {
      smsEnviados++;
      ultimoError = "";
      ultimoEstado = "OK: " + numero;
      drawSuccessScreen(numero, id);
      confirmarProcesado(id);
      delay(1800);
    } else {
      smsFallidos++;
      ultimoError = razonFallo;
      drawErrorSMSScreen(numero, id, razonFallo);
      delay(2500);
    }

    drawHUD();
    delay(300);
  }
}

// ════════════════════════════════════════════════════════
//  ENVIAR SMS — retorna razón de fallo
// ════════════════════════════════════════════════════════
bool enviarSMS(String numero, String mensaje, String &razonFallo) {
  // Validaciones previas
  if (!modemOK) {
    razonFallo = "Modem no iniciado";
    return false;
  }
  if (numero.length() < 8) {
    razonFallo = "Numero invalido";
    return false;
  }
  if (mensaje.length() == 0) {
    razonFallo = "Mensaje vacio";
    return false;
  }
  if (mensaje.length() > 160) {
    log("Advertencia: mensaje >160 chars, puede dividirse.");
  }

  // Enviar comando AT
  Serial1.print("AT+CMGS=\"");
  Serial1.print(numero);
  Serial1.println("\"");
  delay(500);

  // Leer prompt '>'
  String promptResp = "";
  unsigned long tPrompt = millis();
  while (millis() - tPrompt < 3000) {
    while (Serial1.available()) promptResp += (char)Serial1.read();
    if (promptResp.indexOf('>') != -1) break;
  }
  if (promptResp.indexOf('>') == -1) {
    razonFallo = "Sin prompt del modem";
    logError("Modem no envio prompt '>'. Respuesta: " + promptResp);
    return false;
  }

  Serial1.print(mensaje);
  delay(100);
  Serial1.write(26); // Ctrl+Z

  // Esperar OK / ERROR / CMS ERROR
  unsigned long start = millis();
  String resp = "";
  while (millis() - start < 10000) {
    while (Serial1.available()) resp += (char)Serial1.read();

    if (resp.indexOf("OK") != -1) {
      log("Modem respondio OK para " + numero);
      return true;
    }
    if (resp.indexOf("ERROR") != -1) {
      // Detectar tipo de error CMS
      if (resp.indexOf("+CMS ERROR:") != -1) {
        int idx = resp.indexOf("+CMS ERROR:") + 11;
        String cmsCode = resp.substring(idx);
        cmsCode.trim();
        // Solo el código numérico
        int fin = cmsCode.indexOf('\n');
        if (fin != -1) cmsCode = cmsCode.substring(0, fin);
        cmsCode.trim();

        // Traducir códigos CMS comunes
        int code = cmsCode.toInt();
        if      (code == 1)   razonFallo = "Operacion no permitida";
        else if (code == 10)  razonFallo = "SIM no insertada";
        else if (code == 11)  razonFallo = "PIN requerido";
        else if (code == 13)  razonFallo = "SIM falla";
        else if (code == 21)  razonFallo = "Memoria SMS llena";
        else if (code == 28)  razonFallo = "Sin servicio";
        else if (code == 29)  razonFallo = "Error de red";
        else if (code == 38)  razonFallo = "Red no disponible";
        else if (code == 42)  razonFallo = "Red congestionada";
        else if (code == 111) razonFallo = "Numero bloqueado";
        else if (code == 128) razonFallo = "Numero invalido";
        else if (code == 330) razonFallo = "Numero no existe";
        else                  razonFallo = "CMS ERR " + cmsCode;

        logError("CMS ERROR " + cmsCode + ": " + razonFallo);
      } else {
        razonFallo = "ERROR generico modem";
        logError("Respuesta modem: " + resp);
      }
      return false;
    }
  }

  razonFallo = "Timeout (10s sin resp)";
  logError("Timeout esperando respuesta del modem para " + numero);
  return false;
}

// ════════════════════════════════════════════════════════
//  CONFIRMAR PROCESADO
// ════════════════════════════════════════════════════════
void confirmarProcesado(String id) {
  if (WiFi.status() != WL_CONNECTED) {
    logError("Sin WiFi, no se pudo confirmar ID " + id);
    return;
  }
  HTTPClient http;
  String urlConf = String(servidor) + "/api/mensaje/" + id + "/procesado?token=" + token;
  log("Confirmando procesado ID: " + id);
  http.begin(urlConf);
  int code = http.GET();
  log("Confirmacion HTTP: " + String(code));
  http.end();
}
